' =============================================================================
' ИНСТРУКЦИЯ
' 1. Создайте новую книгу Excel и сохраните как .xlsm.
' 2. Alt+F11 → Вставка → Модуль → вставьте ВЕСЬ код из этого файла.
' 3. На листе: Разработчик → Вставить → Кнопка → макрос ZapustitProverku,
'    текст кнопки: «запустить проверку».
' 4. Положите .xlsm в папку с проверяемыми файлами и нажмите кнопку.
'    Файл с макросом не изменяется; обрабатываются только соседние файлы.
'
' ОТЛАДКА: поставьте DEBUG_PROVERKA = True — появится колонка
' «Отладка номер/дата» с причиной «да/нет» по номеру и дате.
' =============================================================================

Option Explicit

Private Const DATA_START_ROW As Long = 6
Private Const HEADER_NAME_ROW As Long = 3
Private Const COL_H As Long = 8          ' номер договора
Private Const COL_I As Long = 9          ' дата договора
Private Const COL_O As Long = 15         ' ФИО контрагента
Private Const COL_W As Long = 23         ' ФИО получателя/отправителя
Private Const COL_Z As Long = 26         ' БИК
Private Const COL_AD As Long = 30        ' содержание операции

Private Const HDR_FIO As String = "ФИО совпадает"
Private Const HDR_NUM_DATE As String = "Номер и Дата есть в содержании"
Private Const HDR_BIK As String = "В БИК только цифры"
Private Const HDR_DEBUG As String = "Отладка номер/дата"

' True = писать колонку с причиной по проверке номера/даты
Private Const DEBUG_PROVERKA As Boolean = False

' -----------------------------------------------------------------------------
' Точка входа для кнопки «запустить проверку»
' -----------------------------------------------------------------------------
Public Sub ZapustitProverku()
    Call ProverkaFailov
End Sub

Public Sub ProverkaFailov()
    Dim folderPath As String
    Dim fileName As String
    Dim thisName As String
    Dim wb As Workbook
    Dim processed As Long
    Dim failed As Long
    Dim errList As String
    Dim ext As String
    Dim fullPath As String
    Dim seen As Object

    folderPath = ThisWorkbook.Path
    If Len(folderPath) = 0 Then
        MsgBox "Сначала сохраните файл с макросом на диск (в нужную папку).", vbExclamation
        Exit Sub
    End If

    thisName = LCase$(ThisWorkbook.Name)
    Set seen = CreateObject("Scripting.Dictionary")
    processed = 0
    failed = 0
    errList = ""

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False

    On Error GoTo CleanFail

    fileName = Dir(folderPath & Application.PathSeparator & "*.xls*")
    Do While Len(fileName) > 0
        If Left$(fileName, 2) <> "~$" Then
            If LCase$(fileName) <> thisName Then
                ext = LCase$(Mid$(fileName, InStrRev(fileName, ".") + 1))
                If ext = "xlsx" Or ext = "xlsm" Then
                    If Not seen.Exists(LCase$(fileName)) Then
                        seen.Add LCase$(fileName), True
                        fullPath = folderPath & Application.PathSeparator & fileName
                        On Error Resume Next
                        Set wb = Workbooks.Open(fileName:=fullPath, UpdateLinks:=0, ReadOnly:=False)
                        If Err.Number <> 0 Or wb Is Nothing Then
                            failed = failed + 1
                            errList = errList & vbCrLf & "- " & fileName & ": " & Err.Description
                            Err.Clear
                        Else
                            On Error GoTo CleanFail
                            ProcessWorksheet wb.Worksheets(1)
                            wb.Save
                            wb.Close SaveChanges:=False
                            Set wb = Nothing
                            processed = processed + 1
                        End If
                        On Error GoTo CleanFail
                    End If
                End If
            End If
        End If
        fileName = Dir()
    Loop

    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True

    MsgBox "Готово." & vbCrLf & _
           "Обработано файлов: " & processed & vbCrLf & _
           "Ошибок открытия: " & failed & _
           IIf(Len(errList) > 0, vbCrLf & errList, ""), _
           vbInformation
    Exit Sub

CleanFail:
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    If Not wb Is Nothing Then
        On Error Resume Next
        wb.Close SaveChanges:=False
        On Error GoTo 0
    End If
    MsgBox "Ошибка: " & Err.Description, vbCritical
End Sub

' -----------------------------------------------------------------------------
' Обработка одного листа
' -----------------------------------------------------------------------------
Private Sub ProcessWorksheet(ByVal ws As Worksheet)
    Dim lastRow As Long
    Dim lastCol As Long
    Dim colFio As Long
    Dim colNumDate As Long
    Dim colBik As Long
    Dim colDebug As Long
    Dim r As Long
    Dim fioO As String
    Dim fioW As String
    Dim contractNo As String
    Dim dateStr As String
    Dim content As String
    Dim bikRaw As String
    Dim hasNum As Boolean
    Dim hasDate As Boolean
    Dim dbg As String

    lastRow = FindLastDataRow(ws)
    If lastRow < DATA_START_ROW Then Exit Sub

    lastCol = FindLastUsedCol(ws)
    colFio = FindOrCreateResultCol(ws, lastCol, HDR_FIO)
    If colFio > lastCol Then lastCol = colFio
    colNumDate = FindOrCreateResultCol(ws, lastCol, HDR_NUM_DATE)
    If colNumDate > lastCol Then lastCol = colNumDate
    colBik = FindOrCreateResultCol(ws, lastCol, HDR_BIK)
    If colBik > lastCol Then lastCol = colBik

    colDebug = 0
    If DEBUG_PROVERKA Then
        colDebug = FindOrCreateResultCol(ws, lastCol, HDR_DEBUG)
    End If

    For r = DATA_START_ROW To lastRow
        ' 1) ФИО: кириллица в памяти; совпадение или вхождение O в W (адрес в W)
        fioO = KeepCyrillic(CStr(ws.Cells(r, COL_O).Value2))
        fioW = KeepCyrillic(CStr(ws.Cells(r, COL_W).Value2))

        If Len(fioO) > 0 And Len(fioW) > 0 And _
           InStr(1, fioW, fioO, vbTextCompare) > 0 Then
            ws.Cells(r, colFio).Value = "да"
        Else
            ws.Cells(r, colFio).Value = "нет"
        End If

        ' 2) номер + дата в AD (ячейки H/I/AD не меняем)
        contractNo = NormalizeSpaces(CStr(ws.Cells(r, COL_H).Value2))
        dateStr = FormatCellDate(ws.Cells(r, COL_I))
        content = NormalizeSpaces(CStr(ws.Cells(r, COL_AD).Value2))

        hasNum = False
        hasDate = False
        dbg = ""

        If Len(contractNo) = 0 Then
            dbg = "H пусто"
        ElseIf Len(dateStr) = 0 Then
            dbg = "I пусто/дата не распознана"
        Else
            hasNum = ContractInContent(content, contractNo)
            hasDate = DateInContent(content, dateStr)
            If hasNum And hasDate Then
                dbg = "ок (номер+дата)"
            Else
                If Not hasNum Then dbg = dbg & "номер не найден; "
                If Not hasDate Then dbg = dbg & "дата не найдена; "
                dbg = dbg & "[H=[" & contractNo & "] I=[" & dateStr & "]]"
            End If
        End If

        If hasNum And hasDate Then
            ws.Cells(r, colNumDate).Value = "да"
        Else
            ws.Cells(r, colNumDate).Value = "нет"
        End If

        If colDebug > 0 Then
            ws.Cells(r, colDebug).Value = dbg
        End If

        ' 3) БИК
        bikRaw = CStr(ws.Cells(r, COL_Z).Value2)
        If IsBikDigitsOnly(bikRaw) Then
            ws.Cells(r, colBik).Value = "да"
        Else
            ws.Cells(r, colBik).Value = "нет"
        End If
    Next r
End Sub

' -----------------------------------------------------------------------------
' Вспомогательные функции
' -----------------------------------------------------------------------------

' Обычные/неразрывные пробелы → " ", Trim по краям
Private Function NormalizeSpaces(ByVal s As String) As String
    Dim t As String
    t = s
    t = Replace(t, ChrW(160), " ")   ' NBSP
    t = Replace(t, ChrW(8239), " ")  ' узкий NBSP
    t = Replace(t, vbTab, " ")
    t = Replace(t, vbCr, " ")
    t = Replace(t, vbLf, " ")
    Do While InStr(t, "  ") > 0
        t = Replace(t, "  ", " ")
    Loop
    NormalizeSpaces = Trim$(t)
End Function

' H="БН" → в AD: «б/н», «бн», «б н», «Б Н» и т.п. (регистр не важен)
' иначе «№» + пробелы + номер (сравнение без учёта регистра)
Private Function ContractInContent(ByVal content As String, ByVal contractNo As String) As Boolean
    Dim p As Long
    Dim i As Long
    Dim re As Object
    Dim extracted As String

    If StrComp(contractNo, "БН", vbTextCompare) = 0 Then
        On Error Resume Next
        Set re = CreateObject("VBScript.RegExp")
        If Not re Is Nothing Then
            re.IgnoreCase = True
            re.Global = False
            ' б/н | бн | б н | Б Н | б / н
            re.Pattern = "б\s*/?\s*н"
            ContractInContent = re.Test(content)
            If Err.Number = 0 Then Exit Function
            Err.Clear
        End If
        On Error GoTo 0
        ' запасной путь без RegExp
        ContractInContent = (InStr(1, content, "б/н", vbTextCompare) > 0) Or _
                            (InStr(1, content, "бн", vbTextCompare) > 0) Or _
                            (InStr(1, content, "б н", vbTextCompare) > 0)
        Exit Function
    End If

    p = 1
    Do
        p = InStr(p, content, "№", vbBinaryCompare)
        If p = 0 Then Exit Do

        i = p + Len("№")
        Do While i <= Len(content) And Mid$(content, i, 1) = " "
            i = i + 1
        Loop

        If i + Len(contractNo) - 1 <= Len(content) Then
            extracted = Mid$(content, i, Len(contractNo))
            If StrComp(extracted, contractNo, vbTextCompare) = 0 Then
                ContractInContent = True
                Exit Function
            End If
        End If
        p = p + 1
    Loop

    ContractInContent = False
End Function

' Дата после «от»: точки/слэши/пробелы, без ведущего нуля, «г»/«г.»
' Примеры: от 06.10.2025г | от 8/09 2025 г | от 08.09.2025
Private Function DateInContent(ByVal content As String, ByVal dateStr As String) As Boolean
    Dim d As Long, m As Long, y As Long
    Dim re As Object
    Dim pat As String

    dateStr = NormalizeSpaces(dateStr)

    ' быстрый путь: точное вхождение
    If InStr(1, content, "от " & dateStr, vbBinaryCompare) > 0 Then
        DateInContent = True
        Exit Function
    End If

    If Not ParseDateParts(dateStr, d, m, y) Then
        DateInContent = False
        Exit Function
    End If

    On Error Resume Next
    Set re = CreateObject("VBScript.RegExp")
    If re Is Nothing Or Err.Number <> 0 Then
        Err.Clear
        On Error GoTo 0
        DateInContent = False
        Exit Function
    End If
    On Error GoTo 0

    re.IgnoreCase = True
    re.Global = False
    ' от + пробелы + день + разделитель + месяц + разделитель + год + опц. г/г.
    pat = "от\s+0*" & CStr(d) & "[\.\/\s]+0*" & CStr(m) & "[\.\/\s]+" & CStr(y) & "\s*г?\.?"
    re.Pattern = pat
    DateInContent = re.Test(content)
End Function

Private Function ParseDateParts(ByVal dateStr As String, ByRef d As Long, ByRef m As Long, ByRef y As Long) As Boolean
    Dim t As String
    Dim i As Long
    Dim ch As String
    Dim buf As String
    Dim arr As Collection

    t = NormalizeSpaces(dateStr)
    If Right$(LCase$(t), 2) = "г." Then t = Left$(t, Len(t) - 2)
    If Right$(LCase$(t), 1) = "г" Then t = Left$(t, Len(t) - 1)
    t = Trim$(t)

    Set arr = New Collection
    buf = ""
    For i = 1 To Len(t)
        ch = Mid$(t, i, 1)
        If ch >= "0" And ch <= "9" Then
            buf = buf & ch
        Else
            If Len(buf) > 0 Then
                arr.Add CLng(buf)
                buf = ""
            End If
        End If
    Next i
    If Len(buf) > 0 Then arr.Add CLng(buf)

    If arr.Count < 3 Then
        ParseDateParts = False
        Exit Function
    End If

    d = CLng(arr(1))
    m = CLng(arr(2))
    y = CLng(arr(3))
    If d < 1 Or d > 31 Or m < 1 Or m > 12 Or y < 1000 Then
        ParseDateParts = False
        Exit Function
    End If
    ParseDateParts = True
End Function

Private Function FindLastDataRow(ByVal ws As Worksheet) As Long
    Dim lr As Long
    lr = ws.Cells(ws.Rows.Count, COL_H).End(xlUp).Row
    If lr < DATA_START_ROW Then
        lr = ws.Cells(ws.Rows.Count, COL_O).End(xlUp).Row
    End If
    FindLastDataRow = lr
End Function

Private Function FindLastUsedCol(ByVal ws As Worksheet) As Long
    Dim lc As Long
    On Error Resume Next
    lc = ws.UsedRange.Columns(ws.UsedRange.Columns.Count).Column
    On Error GoTo 0
    If lc < COL_AD Then lc = COL_AD
    FindLastUsedCol = lc
End Function

Private Function FindOrCreateResultCol(ByVal ws As Worksheet, ByVal lastCol As Long, ByVal headerText As String) As Long
    Dim c As Long
    Dim maxScan As Long

    maxScan = lastCol + 10
    If maxScan < 50 Then maxScan = 50

    For c = 1 To maxScan
        If CStr(ws.Cells(HEADER_NAME_ROW, c).Value2) = headerText Then
            FindOrCreateResultCol = c
            Exit Function
        End If
    Next c

    FindOrCreateResultCol = lastCol + 1
    ws.Cells(HEADER_NAME_ROW, FindOrCreateResultCol).Value = headerText
End Function

Private Function KeepCyrillic(ByVal s As String) As String
    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    result = ""
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        code = AscW(ch)
        If (code >= 1040 And code <= 1103) Or code = 1025 Or code = 1105 Then
            result = result & ch
        End If
    Next i
    KeepCyrillic = result
End Function

Private Function FormatCellDate(ByVal cell As Range) As String
    Dim v As Variant
    Dim t As String
    v = cell.Value2
    If IsEmpty(v) Or v = "" Then
        FormatCellDate = ""
        Exit Function
    End If
    If IsDate(cell.Value) Then
        FormatCellDate = Format$(CDate(cell.Value), "dd.mm.yyyy")
    Else
        t = NormalizeSpaces(CStr(v))
        If Right$(LCase$(t), 2) = "г." Then t = Trim$(Left$(t, Len(t) - 2))
        If Right$(LCase$(t), 1) = "г" Then t = Trim$(Left$(t, Len(t) - 1))
        FormatCellDate = t
    End If
End Function

Private Function IsBikDigitsOnly(ByVal s As String) As Boolean
    Dim t As String
    Dim i As Long
    Dim ch As String

    t = NormalizeSpaces(s)
    If Len(t) = 0 Then
        IsBikDigitsOnly = False
        Exit Function
    End If
    If InStr(t, " ") > 0 Then
        IsBikDigitsOnly = False
        Exit Function
    End If

    For i = 1 To Len(t)
        ch = Mid$(t, i, 1)
        If ch < "0" Or ch > "9" Then
            IsBikDigitsOnly = False
            Exit Function
        End If
    Next i

    IsBikDigitsOnly = True
End Function
