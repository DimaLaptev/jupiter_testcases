' =============================================================================
' ИНСТРУКЦИЯ
' 1. Создайте новую книгу Excel и сохраните как .xlsm.
' 2. Alt+F11 → Вставка → Модуль → вставьте ВЕСЬ код из этого файла.
' 3. На листе: Разработчик → Вставить → Кнопка → макрос ZapustitProverku,
'    текст кнопки: «запустить проверку».
' 4. Положите .xlsm в папку с проверяемыми файлами и нажмите кнопку.
'    Файл с макросом не изменяется; обрабатываются только соседние файлы.
'
' ОТЛАДКА: поставьте DEBUG_PROVERKA = True — появится колонка
' «Отладка номер/дата» с причиной «да/нет» по номеру и дате.
' =============================================================================

Option Explicit

Private Const DATA_START_ROW As Long = 6
Private Const HEADER_NAME_ROW As Long = 3
Private Const COL_H As Long = 8          ' номер договора
Private Const COL_I As Long = 9          ' дата договора
Private Const COL_O As Long = 15         ' ФИО контрагента
Private Const COL_W As Long = 23         ' ФИО получателя/отправителя
Private Const COL_Z As Long = 26         ' БИК
Private Const COL_AD As Long = 30        ' содержание операции

Private Const HDR_FIO As String = "ФИО совпадает"
Private Const HDR_NUM_DATE As String = "Номер и Дата есть в содержании"
Private Const HDR_BIK As String = "В БИК только цифры"
Private Const HDR_DEBUG As String = "Отладка номер/дата"

' True = писать колонку с причиной по проверке номера/даты
Private Const DEBUG_PROVERKA As Boolean = False

' -----------------------------------------------------------------------------
' Точка входа для кнопки «запустить проверку»
' -----------------------------------------------------------------------------
Public Sub ZapustitProverku()
    Call ProverkaFailov
End Sub

Public Sub ProverkaFailov()
    Dim folderPath As String
    Dim fileName As String
    Dim thisName As String
    Dim wb As Workbook
    Dim processed As Long
    Dim failed As Long
    Dim errList As String
    Dim ext As String
    Dim fullPath As String
    Dim seen As Object

    folderPath = ThisWorkbook.Path
    If Len(folderPath) = 0 Then
        MsgBox "Сначала сохраните файл с макросом на диск (в нужную папку).", vbExclamation
        Exit Sub
    End If

    thisName = LCase$(ThisWorkbook.Name)
    Set seen = CreateObject("Scripting.Dictionary")
    processed = 0
    failed = 0
    errList = ""

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False

    On Error GoTo CleanFail

    fileName = Dir(folderPath & Application.PathSeparator & "*.xls*")
    Do While Len(fileName) > 0
        If Left$(fileName, 2) <> "~$" Then
            If LCase$(fileName) <> thisName Then
                ext = LCase$(Mid$(fileName, InStrRev(fileName, ".") + 1))
                If ext = "xlsx" Or ext = "xlsm" Then
                    If Not seen.Exists(LCase$(fileName)) Then
                        seen.Add LCase$(fileName), True
                        fullPath = folderPath & Application.PathSeparator & fileName
                        On Error Resume Next
                        Set wb = Workbooks.Open(fileName:=fullPath, UpdateLinks:=0, ReadOnly:=False)
                        If Err.Number <> 0 Or wb Is Nothing Then
                            failed = failed + 1
                            errList = errList & vbCrLf & "- " & fileName & ": " & Err.Description
                            Err.Clear
                        Else
                            On Error GoTo CleanFail
                            ProcessWorksheet wb.Worksheets(1)
                            wb.Save
                            wb.Close SaveChanges:=False
                            Set wb = Nothing
                            processed = processed + 1
                        End If
                        On Error GoTo CleanFail
                    End If
                End If
            End If
        End If
        fileName = Dir()
    Loop

    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True

    MsgBox "Готово." & vbCrLf & _
           "Обработано файлов: " & processed & vbCrLf & _
           "Ошибок открытия: " & failed & _
           IIf(Len(errList) > 0, vbCrLf & errList, ""), _
           vbInformation
    Exit Sub

CleanFail:
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    If Not wb Is Nothing Then
        On Error Resume Next
        wb.Close SaveChanges:=False
        On Error GoTo 0
    End If
    MsgBox "Ошибка: " & Err.Description, vbCritical
End Sub

' -----------------------------------------------------------------------------
' Обработка одного листа
' -----------------------------------------------------------------------------
Private Sub ProcessWorksheet(ByVal ws As Worksheet)
    Dim lastRow As Long
    Dim lastCol As Long
    Dim colFio As Long
    Dim colNumDate As Long
    Dim colBik As Long
    Dim colDebug As Long
    Dim r As Long
    Dim fioO As String
    Dim fioW As String
    Dim contractNo As String
    Dim dateStr As String
    Dim content As String
    Dim bikRaw As String
    Dim hasNum As Boolean
    Dim hasDate As Boolean
    Dim dbg As String

    lastRow = FindLastDataRow(ws)
    If lastRow < DATA_START_ROW Then Exit Sub

    lastCol = FindLastUsedCol(ws)
    colFio = FindOrCreateResultCol(ws, lastCol, HDR_FIO)
    If colFio > lastCol Then lastCol = colFio
    colNumDate = FindOrCreateResultCol(ws, lastCol, HDR_NUM_DATE)
    If colNumDate > lastCol Then lastCol = colNumDate
    colBik = FindOrCreateResultCol(ws, lastCol, HDR_BIK)
    If colBik > lastCol Then lastCol = colBik

    colDebug = 0
    If DEBUG_PROVERKA Then
        colDebug = FindOrCreateResultCol(ws, lastCol, HDR_DEBUG)
    End If

    For r = DATA_START_ROW To lastRow
        ' 1) ФИО: кириллица в памяти; совпадение или вхождение O в W (адрес в W)
        fioO = KeepCyrillic(CStr(ws.Cells(r, COL_O).Value2))
        fioW = KeepCyrillic(CStr(ws.Cells(r, COL_W).Value2))

        If Len(fioO) > 0 And Len(fioW) > 0 And _
           InStr(1, fioW, fioO, vbTextCompare) > 0 Then
            ws.Cells(r, colFio).Value = "да"
        Else
            ws.Cells(r, colFio).Value = "нет"
        End If

        ' 2) номер + дата в AD (ячейки H/I/AD не меняем)
        contractNo = NormalizeSpaces(CStr(ws.Cells(r, COL_H).Value2))
        dateStr = FormatCellDate(ws.Cells(r, COL_I))
        content = NormalizeSpaces(CStr(ws.Cells(r, COL_AD).Value2))

        hasNum = False
        hasDate = False
        dbg = ""

        If Len(contractNo) = 0 Then
            dbg = "H пусто"
        ElseIf Len(dateStr) = 0 Then
            dbg = "I пусто/дата не распознана"
        Else
            hasNum = ContractInContent(content, contractNo)
            hasDate = DateInContent(content, dateStr)
            ' запасной шаблон: «<номер> от <дата>» подтверждает оба условия
            If (Not hasNum Or Not hasDate) And _
               ContractDatePairInContent(content, contractNo, dateStr) Then
                hasNum = True
                hasDate = True
            End If
            If hasNum And hasDate Then
                dbg = "ок (номер+дата)"
            Else
                If Not hasNum Then dbg = dbg & "номер не найден; "
                If Not hasDate Then dbg = dbg & "дата не найдена; "
                dbg = dbg & "[H=[" & contractNo & "] I=[" & dateStr & "]]"
            End If
        End If

        If hasNum And hasDate Then
            ws.Cells(r, colNumDate).Value = "да"
        Else
            ws.Cells(r, colNumDate).Value = "нет"
        End If

        If colDebug > 0 Then
            ws.Cells(r, colDebug).Value = dbg
        End If

        ' 3) БИК
        bikRaw = CStr(ws.Cells(r, COL_Z).Value2)
        If IsBikDigitsOnly(bikRaw) Then
            ws.Cells(r, colBik).Value = "да"
        Else
            ws.Cells(r, colBik).Value = "нет"
        End If
    Next r
End Sub

' Вспомогательные функции
' -----------------------------------------------------------------------------

' Обычные/неразрывные пробелы → " ", Trim по краям
Private Function NormalizeSpaces(ByVal s As String) As String
    Dim t As String
    t = s
    t = Replace(t, ChrW(160), " ")   ' NBSP
    t = Replace(t, ChrW(8239), " ")  ' узкий NBSP
    t = Replace(t, vbTab, " ")
    t = Replace(t, vbCr, " ")
    t = Replace(t, vbLf, " ")
    Do While InStr(t, "  ") > 0
        t = Replace(t, "  ", " ")
    Loop
    NormalizeSpaces = Trim$(t)
End Function

' H="БН" → «б/н»…, «ТД от…», «дог. от…» / «дог. б/н»;
' иначе: «№»/«договору»/«ТД»/«дог.» + номер
Private Function ContractInContent(ByVal content As String, ByVal contractNo As String) As Boolean
    If StrComp(contractNo, "БН", vbTextCompare) = 0 Then
        ' б/н-варианты, «ТД от», «дог. от», «дог. б/н»
        If BnMarkerInContent(content) Then
            ContractInContent = True
            Exit Function
        End If
        ContractInContent = False
        Exit Function
    End If

    If PrefixThenContract(content, "№", contractNo) Then
        ContractInContent = True
        Exit Function
    End If
    If PrefixThenContract(content, "договору", contractNo) Then
        ContractInContent = True
        Exit Function
    End If
    If PrefixThenContract(content, "ТД", contractNo) Then
        ContractInContent = True
        Exit Function
    End If
    If PrefixThenContract(content, "дог.", contractNo) Then
        ContractInContent = True
        Exit Function
    End If

    ContractInContent = False
End Function

' Маркеры договора без номера в AD
Private Function BnMarkerInContent(ByVal content As String) As Boolean
    Dim re As Object
    Dim p As Long
    Dim i As Long

    On Error Resume Next
    Set re = CreateObject("VBScript.RegExp")
    If Not re Is Nothing And Err.Number = 0 Then
        re.IgnoreCase = True
        re.Global = False
        re.Pattern = "б\s*/?\s*н"
        If re.Test(content) Then
            BnMarkerInContent = True
            Exit Function
        End If
    End If
    Err.Clear
    On Error GoTo 0

    If InStr(1, content, "б/н", vbTextCompare) > 0 Or _
       InStr(1, content, "бн", vbTextCompare) > 0 Or _
       InStr(1, content, "б н", vbTextCompare) > 0 Then
        BnMarkerInContent = True
        Exit Function
    End If

    ' «ТД от …» без номера между ТД и от
    p = 1
    Do
        p = FindTokenPos(content, "ТД", p)
        If p = 0 Then Exit Do
        i = p + Len("ТД")
        Do While i <= Len(content) And Mid$(content, i, 1) = " "
            i = i + 1
        Loop
        If i + 1 <= Len(content) Then
            If StrComp(Mid$(content, i, 2), "от", vbTextCompare) = 0 Then
                BnMarkerInContent = True
                Exit Function
            End If
        End If
        p = p + 1
    Loop

    ' «дог. от …» или «дог. б/н»
    p = 1
    Do
        p = InStr(p, content, "дог.", vbTextCompare)
        If p = 0 Then Exit Do
        i = p + Len("дог.")
        Do While i <= Len(content) And Mid$(content, i, 1) = " "
            i = i + 1
        Loop
        If i + 1 <= Len(content) Then
            If StrComp(Mid$(content, i, 2), "от", vbTextCompare) = 0 Then
                BnMarkerInContent = True
                Exit Function
            End If
        End If
        If HasTokenAt(content, i, "б/н") Or HasTokenAt(content, i, "бн") Or HasTokenAt(content, i, "БН") Then
            BnMarkerInContent = True
            Exit Function
        End If
        p = p + 1
    Loop

    BnMarkerInContent = False
End Function

' Префикс (№ / договору / ТД / дог.) + пробелы + опционально «№» + номер
Private Function PrefixThenContract(ByVal content As String, ByVal prefix As String, ByVal contractNo As String) As Boolean
    Dim p As Long
    Dim i As Long
    Dim needTokenBoundary As Boolean

    needTokenBoundary = (StrComp(prefix, "ТД", vbTextCompare) = 0)

    p = 1
    Do
        If needTokenBoundary Then
            p = FindTokenPos(content, prefix, p)
        Else
            p = InStr(p, content, prefix, vbTextCompare)
        End If
        If p = 0 Then Exit Do

        i = p + Len(prefix)
        Do While i <= Len(content) And Mid$(content, i, 1) = " "
            i = i + 1
        Loop

        ' «ТД № 22» / «дог. № 22» — пропускаем № между префиксом и номером
        If Mid$(content, i, 1) = "№" Then
            i = i + 1
            Do While i <= Len(content) And Mid$(content, i, 1) = " "
                i = i + 1
            Loop
        End If

        If HasTokenAt(content, i, contractNo) Then
            PrefixThenContract = True
            Exit Function
        End If
        p = p + 1
    Loop

    PrefixThenContract = False
End Function

' Шаблон: «<номер> от <дата>»; для БН также «ТД от <дата>» / «дог. от <дата>»
Private Function ContractDatePairInContent(ByVal content As String, ByVal contractNo As String, ByVal dateStr As String) As Boolean
    Dim p As Long
    Dim i As Long
    Dim after As String
    Dim prefixes As Variant
    Dim pref As Variant

    If StrComp(contractNo, "БН", vbTextCompare) = 0 Then
        prefixes = Array("ТД", "дог.")
        For Each pref In prefixes
            p = 1
            Do
                If StrComp(CStr(pref), "ТД", vbTextCompare) = 0 Then
                    p = FindTokenPos(content, CStr(pref), p)
                Else
                    p = InStr(p, content, CStr(pref), vbTextCompare)
                End If
                If p = 0 Then Exit Do

                i = p + Len(CStr(pref))
                Do While i <= Len(content) And Mid$(content, i, 1) = " "
                    i = i + 1
                Loop

                If i + 1 <= Len(content) Then
                    If StrComp(Mid$(content, i, 2), "от", vbTextCompare) = 0 Then
                        i = i + 2
                        Do While i <= Len(content) And Mid$(content, i, 1) = " "
                            i = i + 1
                        Loop
                        If DateStartsAt(Mid$(content, i), dateStr) Then
                            ContractDatePairInContent = True
                            Exit Function
                        End If
                    End If
                End If
                p = p + 1
            Loop
        Next pref
        ContractDatePairInContent = False
        Exit Function
    End If

    p = 1
    Do
        p = FindTokenPos(content, contractNo, p)
        If p = 0 Then Exit Do

        i = p + Len(contractNo)
        Do While i <= Len(content) And Mid$(content, i, 1) = " "
            i = i + 1
        Loop

        If i + 1 <= Len(content) Then
            If StrComp(Mid$(content, i, 2), "от", vbTextCompare) = 0 Then
                i = i + 2
                Do While i <= Len(content) And Mid$(content, i, 1) = " "
                    i = i + 1
                Loop
                after = Mid$(content, i)
                If DateStartsAt(after, dateStr) Then
                    ContractDatePairInContent = True
                    Exit Function
                End If
            End If
        End If
        p = p + 1
    Loop

    ContractDatePairInContent = False
End Function

' True, если в позиции pos стоит token и по краям граница (не продолжение 14→140)
Private Function HasTokenAt(ByVal content As String, ByVal pos As Long, ByVal token As String) As Boolean
    If pos < 1 Or pos + Len(token) - 1 > Len(content) Then
        HasTokenAt = False
        Exit Function
    End If
    If StrComp(Mid$(content, pos, Len(token)), token, vbTextCompare) <> 0 Then
        HasTokenAt = False
        Exit Function
    End If
    If pos > 1 Then
        If Not IsTokenBoundaryChar(Mid$(content, pos - 1, 1)) Then
            HasTokenAt = False
            Exit Function
        End If
    End If
    If pos + Len(token) <= Len(content) Then
        If Not IsTokenBoundaryChar(Mid$(content, pos + Len(token), 1)) Then
            HasTokenAt = False
            Exit Function
        End If
    End If
    HasTokenAt = True
End Function

Private Function FindTokenPos(ByVal content As String, ByVal token As String, ByVal startPos As Long) As Long
    Dim p As Long
    p = startPos
    If p < 1 Then p = 1
    Do
        p = InStr(p, content, token, vbTextCompare)
        If p = 0 Then
            FindTokenPos = 0
            Exit Function
        End If
        If HasTokenAt(content, p, token) Then
            FindTokenPos = p
            Exit Function
        End If
        p = p + 1
    Loop
End Function

Private Function IsTokenBoundaryChar(ByVal ch As String) As Boolean
    Dim code As Long
    If Len(ch) = 0 Then
        IsTokenBoundaryChar = True
        Exit Function
    End If
    If ch >= "0" And ch <= "9" Then
        IsTokenBoundaryChar = False
        Exit Function
    End If
    code = AscW(ch)
    ' A-Z a-z
    If (code >= 65 And code <= 90) Or (code >= 97 And code <= 122) Then
        IsTokenBoundaryChar = False
        Exit Function
    End If
    ' А-Я а-я Ё ё
    If (code >= 1040 And code <= 1103) Or code = 1025 Or code = 1105 Then
        IsTokenBoundaryChar = False
        Exit Function
    End If
    IsTokenBoundaryChar = True
End Function

' Проверка, что строка s начинается с даты dateStr (дд.мм.гггг / дд.мм.гг / дд.мм.гг.)
Private Function DateStartsAt(ByVal s As String, ByVal dateStr As String) As Boolean
    Dim d As Long, m As Long, y As Long
    Dim re As Object
    Dim pat As String
    Dim shortDt As String

    dateStr = NormalizeSpaces(dateStr)
    s = NormalizeSpaces(s)

    If Len(dateStr) > 0 And Left$(s, Len(dateStr)) = dateStr Then
        DateStartsAt = True
        Exit Function
    End If

    If Not ParseDateParts(dateStr, d, m, y) Then
        DateStartsAt = False
        Exit Function
    End If

    shortDt = DateShortDotted(d, m, y)
    If Left$(s, Len(shortDt)) = shortDt Then
        DateStartsAt = True
        Exit Function
    End If
    If Left$(s, Len(shortDt) + 1) = shortDt & "." Then
        DateStartsAt = True
        Exit Function
    End If

    On Error Resume Next
    Set re = CreateObject("VBScript.RegExp")
    If re Is Nothing Or Err.Number <> 0 Then
        Err.Clear
        On Error GoTo 0
        DateStartsAt = False
        Exit Function
    End If
    On Error GoTo 0

    re.IgnoreCase = True
    re.Global = False
    ' год: 2025 | 25; после года — точка, «г», пробел или конец (не цифра)
    pat = "^0*" & CStr(d) & "[\.\/\s]+0*" & CStr(m) & "[\.\/\s]+" & YearPattern(y) & "(\.|г\.?|[\s,;)]|$)"
    re.Pattern = pat
    DateStartsAt = re.Test(s)
End Function

' Дата после «от»: дд.мм.гггг ↔ дд.мм.гг / дд.мм.гг.
Private Function DateInContent(ByVal content As String, ByVal dateStr As String) As Boolean
    Dim d As Long, m As Long, y As Long
    Dim re As Object
    Dim pat As String
    Dim shortDt As String

    dateStr = NormalizeSpaces(dateStr)

    If InStr(1, content, "от " & dateStr, vbBinaryCompare) > 0 Then
        DateInContent = True
        Exit Function
    End If

    If Not ParseDateParts(dateStr, d, m, y) Then
        DateInContent = False
        Exit Function
    End If

    ' явный короткий год: от 02.06.25 / от 02.06.25.
    shortDt = DateShortDotted(d, m, y)
    If InStr(1, content, "от " & shortDt, vbBinaryCompare) > 0 Then
        DateInContent = True
        Exit Function
    End If

    On Error Resume Next
    Set re = CreateObject("VBScript.RegExp")
    If re Is Nothing Or Err.Number <> 0 Then
        Err.Clear
        On Error GoTo 0
        DateInContent = False
        Exit Function
    End If
    On Error GoTo 0

    re.IgnoreCase = True
    re.Global = False
    pat = "от\s+0*" & CStr(d) & "[\.\/\s]+0*" & CStr(m) & "[\.\/\s]+" & YearPattern(y) & "(\.|г\.?|[\s,;)]|$)"
    re.Pattern = pat
    DateInContent = re.Test(content)
End Function

' 02.06.25
Private Function DateShortDotted(ByVal d As Long, ByVal m As Long, ByVal y As Long) As String
    DateShortDotted = Right$("0" & CStr(d), 2) & "." & _
                      Right$("0" & CStr(m), 2) & "." & _
                      Right$("0" & CStr(y Mod 100), 2)
End Function

' Полный год или две последние цифры: 2025 | 25
Private Function YearPattern(ByVal y As Long) As String
    YearPattern = "(" & CStr(y) & "|" & Right$("0" & CStr(y Mod 100), 2) & ")"
End Function

Private Function ParseDateParts(ByVal dateStr As String, ByRef d As Long, ByRef m As Long, ByRef y As Long) As Boolean
    Dim t As String
    Dim i As Long
    Dim ch As String
    Dim buf As String
    Dim arr As Collection

    t = NormalizeSpaces(dateStr)
    If Right$(LCase$(t), 2) = "г." Then t = Left$(t, Len(t) - 2)
    If Right$(LCase$(t), 1) = "г" Then t = Left$(t, Len(t) - 1)
    t = Trim$(t)

    Set arr = New Collection
    buf = ""
    For i = 1 To Len(t)
        ch = Mid$(t, i, 1)
        If ch >= "0" And ch <= "9" Then
            buf = buf & ch
        Else
            If Len(buf) > 0 Then
                arr.Add CLng(buf)
                buf = ""
            End If
        End If
    Next i
    If Len(buf) > 0 Then arr.Add CLng(buf)

    If arr.Count < 3 Then
        ParseDateParts = False
        Exit Function
    End If

    d = CLng(arr(1))
    m = CLng(arr(2))
    y = CLng(arr(3))
    If d < 1 Or d > 31 Or m < 1 Or m > 12 Or y < 1000 Then
        ParseDateParts = False
        Exit Function
    End If
    ParseDateParts = True
End Function

Private Function FindLastDataRow(ByVal ws As Worksheet) As Long
    Dim lr As Long
    lr = ws.Cells(ws.Rows.Count, COL_H).End(xlUp).Row
    If lr < DATA_START_ROW Then
        lr = ws.Cells(ws.Rows.Count, COL_O).End(xlUp).Row
    End If
    FindLastDataRow = lr
End Function

Private Function FindLastUsedCol(ByVal ws As Worksheet) As Long
    Dim lc As Long
    On Error Resume Next
    lc = ws.UsedRange.Columns(ws.UsedRange.Columns.Count).Column
    On Error GoTo 0
    If lc < COL_AD Then lc = COL_AD
    FindLastUsedCol = lc
End Function

Private Function FindOrCreateResultCol(ByVal ws As Worksheet, ByVal lastCol As Long, ByVal headerText As String) As Long
    Dim c As Long
    Dim maxScan As Long

    maxScan = lastCol + 10
    If maxScan < 50 Then maxScan = 50

    For c = 1 To maxScan
        If CStr(ws.Cells(HEADER_NAME_ROW, c).Value2) = headerText Then
            FindOrCreateResultCol = c
            Exit Function
        End If
    Next c

    FindOrCreateResultCol = lastCol + 1
    ws.Cells(HEADER_NAME_ROW, FindOrCreateResultCol).Value = headerText
End Function

Private Function KeepCyrillic(ByVal s As String) As String
    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    result = ""
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        code = AscW(ch)
        If (code >= 1040 And code <= 1103) Or code = 1025 Or code = 1105 Then
            result = result & ch
        End If
    Next i
    KeepCyrillic = result
End Function

Private Function FormatCellDate(ByVal cell As Range) As String
    Dim v As Variant
    Dim t As String
    v = cell.Value2
    If IsEmpty(v) Or v = "" Then
        FormatCellDate = ""
        Exit Function
    End If
    If IsDate(cell.Value) Then
        FormatCellDate = Format$(CDate(cell.Value), "dd.mm.yyyy")
    Else
        t = NormalizeSpaces(CStr(v))
        If Right$(LCase$(t), 2) = "г." Then t = Trim$(Left$(t, Len(t) - 2))
        If Right$(LCase$(t), 1) = "г" Then t = Trim$(Left$(t, Len(t) - 1))
        FormatCellDate = t
    End If
End Function

Private Function IsBikDigitsOnly(ByVal s As String) As Boolean
    Dim t As String
    Dim i As Long
    Dim ch As String

    t = NormalizeSpaces(s)
    If Len(t) = 0 Then
        IsBikDigitsOnly = False
        Exit Function
    End If
    If InStr(t, " ") > 0 Then
        IsBikDigitsOnly = False
        Exit Function
    End If

    For i = 1 To Len(t)
        ch = Mid$(t, i, 1)
        If ch < "0" Or ch > "9" Then
            IsBikDigitsOnly = False
            Exit Function
        End If
    Next i

    IsBikDigitsOnly = True
End Function
